```
// CompUnit    ::= {Decl | FuncDef};
// FuncDef     ::= FuncType IDENT "(" [FuncFParams] ")" Block;
// FuncType    ::= "int" | "void";
// FuncFParams ::= FuncFParam {"," FuncFParam};
// FuncFParam  ::= BType IDENT | BType IDENT "[" "]";
// Block       ::= "{" {BlockItem} "}";
// BlockItem   ::= Decl | Stmt;
// Decl        ::= ConstDecl | VarDecl;
// ConstDecl   ::= "const" BType ConstDef {"," ConstDef} ";";
// BType       ::= "int";
// ConstDef    ::= IDENT "=" ConstInitVal | IDENT "[" ConstExp "]" "=" ConstInitVal;
// ConstInitVal::= ConstExp | "{" [ConstInitList] "}";
// ConstInitList::= ConstInitVal {"," ConstInitVal};
// VarDecl     ::= BType VarDef {"," VarDef} ";";
// VarDef      ::= IDENT | IDENT "=" InitVal | IDENT "[" ConstExp "]" | IDENT "[" ConstExp "]" "=" InitVal;
// InitVal     ::= Exp | "{" [InitList] "}";
// InitList    ::= InitVal {"," InitVal};
// Stmt        ::= LVal "=" Exp ";" | "return" [Exp] ";" | Block | "if" "(" Exp ")" Stmt ["else" Stmt] | "while" "(" Exp ")" Stmt | "break" ";" | "continue" ";" | [Exp] ";";
// Exp         ::= LOrExp;
// LVal        ::= IDENT | IDENT "[" Exp "]" | LVal "[" Exp "]";
// PrimaryExp  ::= "(" Exp ")" | LVal | Number;
// Number      ::= INT_CONST;
// UnaryExp    ::= PrimaryExp | UnaryOp UnaryExp | IDENT "(" [FuncRParams] ")";
// UnaryOp     ::= "+" | "-" | "!";
// FuncRParams ::= Exp {"," Exp};
// MulExp      ::= UnaryExp | MulExp ("*" | "/" | "%") UnaryExp;
// AddExp      ::= MulExp | AddExp ("+" | "-") MulExp;
// RelExp      ::= AddExp | RelExp ("<" | ">" | "<=" | ">=") AddExp;
// EqExp       ::= RelExp | EqExp ("==" | "!=") RelExp;
// LAndExp     ::= EqExp | LAndExp "&&" EqExp;
// LOrExp      ::= LAndExp | LOrExp "||" LAndExp;
// ConstExp    ::= Exp;
```

```
运算符的优先级（越往下优先级越高）：
LOrExp (逻辑或)：优先级最低。||
LAndExp (逻辑与)：&&
EqExp (相等性判断)：==, !=
RelExp (关系运算)：<, >, <=, >=
AddExp (加减法)：+, -
MulExp (乘除法)：*, /, %
UnaryExp (单目运算)：!, -, +
函数调用：IDENT()
数组下标：[]
优先级最高。
```

```
Lv9 新增语法说明：
1. ConstDef 新增数组形式：IDENT "[" ConstExp "]" "=" ConstInitVal
2. VarDef 新增数组形式：IDENT "[" ConstExp "]" 和 IDENT "[" ConstExp "]" "=" InitVal
3. LVal 新增下标访问：IDENT "[" Exp "]" 和 LVal "[" Exp "]"（支持多维数组）
4. FuncFParam 新增数组参数：BType IDENT "[" "]"（第一维省略）
5. ConstInitVal 新增初始化列表："{" [ConstInitList] "}"
6. InitVal 新增初始化列表："{" [InitList] "}"
7. UnaryExp 新增函数调用：IDENT "(" [FuncRParams] ")"
8. 新增 ConstInitList 和 InitList 规则处理逗号分隔的初始化元素列表
```

```
数组相关语法详解：

1. 数组声明与定义：
   - 一维数组：int arr[10];
   - 一维数组初始化：int arr[5] = {1, 2, 3};
   - 多维数组：int matrix[2][3];
   - 多维数组初始化：int matrix[2][3] = {{1, 2, 3}, {4, 5, 6}};

2. 数组元素访问：
   - 一维数组：arr[i]
   - 多维数组：matrix[i][j]

3. 数组作为函数参数：
   - void func(int arr[]);
   - void func(int arr[], int size);

4. 初始化列表规则：
   - 空列表：{}
   - 单元素：{1}
   - 多元素：{1, 2, 3}
   - 嵌套列表（多维数组）：{{1, 2}, {3, 4}}

5. 数组元素地址计算（行优先存储）：
   - 对于 arr[i][j][k]（维度为 dim1 × dim2 × dim3）：
     - 偏移 = i × dim2 × dim3 × 4 + j × dim3 × 4 + k × 4
     - 其中 4 是 int 类型的字节大小
```
